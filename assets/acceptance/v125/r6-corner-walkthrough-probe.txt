逐条解读（与账目对照）：

① 触发几何——**在「快要看见墙后」的几何瞬间，滞后 1.4cm/零帧**。
   显形线（K(8.05,-26.8)–R(7.2,-27.0) 连线）在巷轴 x=9.3 的理论跨点
   z=-26.506。细步实录：z=-26.46 时 cross=+0.039（差 3.9cm 未跨），
   下一帧步进到 z=-26.52（过线 1.4cm，cross=-0.012）**同一渲染帧
   引爆**（STEP-TRIGGER 即 sub=reveal clk=0）。真机 60fps × 4.2m/s
   单帧步长 7cm——触发粒度就是一帧脚步，不提前不拖延。与
   cornerscare.test 显形线几何账零矛盾。

② 镜头特写——**错拍开始前已死锁对准，全程跟焦**。触发帧 aim
   61.9°（玩家行走朝向与魅影夹角）→ 0.1 游戏秒后 19.7° → 0.2
   游戏秒后 0.8° → stare 拍起点 **0.0°**（每帧 lookAt 跟焦，直到
   blackout 归还）。FOV 慢推 69.8→69.3→68.4→67.3→66→64.6→63.1
   （smoothstep：clk0.607 时 pRaw=0.4→push≈0.35→70−15×0.35≈64.7≈
   实测 64.6，CLOSEUP.fovPush=15 账目吻合）；pitch 0.027→0.295
   仰角跟焦（它 1.97m 高、最近 0.95m）。双脚全程钉死 (9.3,-26.52)。

③ 拍点实时钟——later 链逐拍吻合 SCARE_BEATS：stare 实测 +0.82s
   （账 550ms+帧界）/ rush +1.54s（账 1500ms）/ shock 的 uShock
   在 rush 中段亮起 0.82 并指数衰减（账 1900ms 后）/ blackout
   +2.4~2.6s 魅影隐藏+rim 归零+FOV 归还 70（账 2400ms）/ wake
   +3.3s 移回 (9.7,9.5) yaw 0 pitch −0.36 俯冲醒（账 3300ms +
   WAKE_DAZE）。rim 灯语按拍走：reveal 打火 4.83 → stare 呼吸
   2.5~6.9 → rush 涌光 7.35→8.2 → blackout 归零（RIM_BEATS 吻合）。

④ 软渲染已知错配（非病灶，真机不存在）：later 链走实时钟、滑出/
   扑近/入锁动画走游戏钟——本跑游戏钟约 0.28 倍速（clk 0.807 /
   2.9 实秒），故 rush 拍魅影只从 dist 1.09 逼近到 0.95 就到黑幕。
   真机 60fps 两钟同速：0.55s 滑出全程、0.4s 冲到脸前 0.85m、
   0.35s 入锁全部按账走。第 5 轮 0.065 倍速互证口径复认。

结论：**零矛盾、零病灶——拐角惊吓四层零改动维持**。「吓不吓人」
的体感命题仍等用户真机证词（触发时机与镜头机械层本轮已证到位）。

附录：B 跑探针源码（SV_SHOT_PRE 注入，复现用；A 跑探针即本探针去掉
细步块、换 50ms setInterval——已证不可用，留此为口径）：

(() => {
  const S = window.__SV__;
  if (!S || !S.hall || S.hall() !== 'mulholland') return 'r6:skip-hall';
  if (window.__R6__) return 'r6:dup';
  window.__R6__ = true;
  const K = { x: 8.05, z: -26.8 }, R = { x: 7.2, z: -27.0 };
  const lx = K.x - R.x, lz = K.z - R.z;
  const cam = S.engine.camera;
  const pitchObj = cam.parent;
  let wraith = null, rim = null;
  S.engine.scene.traverse((o) => {
    if (!wraith && o.userData && o.userData.setLurch && o.userData.setRush) wraith = o;
    if (!rim && o.isPointLight && Math.abs(o.position.x - 6.3) < 0.01 && Math.abs(o.position.z + 28.7) < 0.01) rim = o;
  });
  const crossOf = (p) => +(lx * (p.z - K.z) - lz * (p.x - K.x)).toFixed(4);
  const va = wraith.position.clone(), vb = wraith.position.clone(), vc = wraith.position.clone();
  const t0 = Date.now();
  let frame = 0;
  const st = { z: -25.8, stepping: true, fired: false, postWake: 0, done: false };
  S.engine.onUpdate((dt) => {
    frame++;
    if (st.done) return;
    const s = S.scareProbe();
    const p = S.player();
    const cross = crossOf(p);
    if (st.stepping) {
      if (s && s.phase === 2) {
        st.stepping = false;
        st.fired = true;
        console.log('[sv] glb-r6 STEP-TRIGGER ' + JSON.stringify({
          at: [+p.x.toFixed(3), +p.z.toFixed(3)], cross, yaw: +p.yaw.toFixed(3), sub: s.sub, clk: s.clock
        }));
      } else if (st.z < -27.4) {
        st.stepping = false;
        st.done = true;
        console.log('[sv] glb-r6 STEP-EXHAUSTED phase=' + (s ? s.phase : -1));
        return;
      } else {
        console.log('[sv] glb-r6 STEP ' + JSON.stringify({
          prev: [+p.x.toFixed(3), +p.z.toFixed(3)], prevCross: cross, ph: s ? s.phase : -1, to: +st.z.toFixed(3)
        }));
        S.teleport(9.3, st.z, 0);
        st.z -= 0.06;
        return;
      }
    }
    if (!st.fired) return;
    cam.getWorldPosition(va);
    cam.getWorldDirection(vb);
    let aim = null, dist = null;
    if (wraith.visible) {
      vc.set(wraith.position.x - va.x, 1.97 - va.y, wraith.position.z - va.z);
      dist = +Math.hypot(wraith.position.x - p.x, wraith.position.z - p.z).toFixed(2);
      aim = +(Math.acos(Math.max(-1, Math.min(1, vb.dot(vc) / (vb.length() * vc.length())))) * 180 / Math.PI).toFixed(1);
    }
    console.log('[sv] glb-r6 FRAME ' + JSON.stringify({
      F: frame, T: Date.now() - t0, dt: +dt.toFixed(3),
      p: [+p.x.toFixed(2), +p.z.toFixed(2)], yaw: +p.yaw.toFixed(3),
      pitch: +pitchObj.rotation.x.toFixed(3), fov: +cam.fov.toFixed(1),
      ph: s ? s.phase : -1, sub: s ? s.sub : null, clk: s ? s.clock : null,
      w: wraith.visible ? [+wraith.position.x.toFixed(2), +wraith.position.z.toFixed(2)] : 0,
      aim, dist,
      rim: rim ? +rim.intensity.toFixed(2) : null,
      shock: +S.engine.lynchPass.uniforms.uShock.value.toFixed(2)
    }));
    if (s && s.phase === 0 && Math.abs(p.x - 9.7) < 0.05 && Math.abs(p.z - 9.5) < 0.05) {
      st.postWake++;
      if (st.postWake >= 6) {
        st.done = true;
        console.log('[sv] glb-r6 DONE frames=' + frame);
      }
    }
  });
  return 'r6:probeB-armed wraith=' + (wraith ? 'found' : 'MISS') + ' rim=' + (rim ? 'found' : 'MISS');
})()
